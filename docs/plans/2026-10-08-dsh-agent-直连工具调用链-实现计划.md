# DSH Agent 直连工具调用链实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将 AT Terminal 从基于外部 MCP Hub 跨进程中转架构重构为 DSH（DeepSeek Harness）原生工具直接调用体系，使 DSH Agent 可在会话内直接高效调用 11 个 SSH/SFTP 运维工具，同时完整承接并保持既有的三级信任评估（`none`/`policy`/`full`）、命令 AST 安全沙盒、SFTP 目录授权与审计机制。

**Architecture:** 
1. 解耦 `src/agent/` 的交互确认逻辑，抽象 `AgentConfirmationPort` 并在 VS Code 与 DSH 两侧分别适配，消除 `src/agent/` 对 `vscode` 的强耦合；
2. 在 `src/platform/dsh/` 中装配 Host 侧 `AgentToolService` 并导出至 `dsh/host.bundle.js`；
3. 在 `dsh/tools.js` 中将 11 个运维工具规范化定义为符合 DSH 严格 JSON Schema 契约的原生工具，支持协作式取消（`exec.signal`）与富文本渲染（`output.render`）；
4. 在 `dsh/index.js` 中通过生命周期安全的方式挂载到 `ctx.tools`，并接入 DSH 原生审批流（`ctx.approval`），实现毫秒级就地安全分发。

**Tech Stack:** TypeScript, Node.js (v24), Cordis Context/Events, `@deepseek-ai/dsh-tools`, `@deepseek-ai/schemastery`, ssh2, esbuild, vitest.

```ts
import * as vscode from 'vscode';
import type { AgentConfirmationPort, CommandConfirmationRequest } from '../../agent/AgentConfirmationPort';
import type { SftpWriteConfirmation, SftpWriteScope } from '../../agent/SftpWriteAuthorizer';
import { t } from '../../i18n/t';
import { formatRemoteCommandConfirmMessage } from '../../utils/commandPreview';

const CONFIRMATION_TIMEOUT_MS = 120_000;

export class VscodeAgentConfirmation implements AgentConfirmationPort {
  async confirmCommand(request: CommandConfirmationRequest): Promise<boolean> {
    const runCommandLabel = t('Run Command');
    const confirmation = Promise.resolve(
      vscode.window.showWarningMessage(
        formatRemoteCommandConfirmMessage({
          serverLabel: request.serverLabel,
          host: request.host,
          command: request.command,
          destructive: request.destructive,
          riskSummaries: request.riskSummaries
        }),
        { modal: true },
        runCommandLabel
      )
    );

    let timer: ReturnType<typeof setTimeout> | undefined;
    const timeoutPromise = new Promise<never>((_, reject) => {
      timer = setTimeout(() => {
        reject(new Error('Confirmation timed out; ask the user to approve the command dialog in the IDE, then retry.'));
      }, CONFIRMATION_TIMEOUT_MS);
    });

    try {
      const answer = await Promise.race([confirmation, timeoutPromise]);
      return answer === runCommandLabel;
    } catch (error) {
      void confirmation.then((lateAnswer) => {
        if (lateAnswer === runCommandLabel) {
          void vscode.window.showInformationMessage(
            t('The approval arrived after the request timed out — the command was not run. Ask the agent to retry.')
          );
        }
      });
      throw error;
    } finally {
      if (timer) clearTimeout(timer);
    }
  }

  async confirmSftpWrite(confirmation: SftpWriteConfirmation): Promise<SftpWriteScope | undefined> {
    // 迁移原 SftpWriteAuthorizer.ts 中的 confirmWithVscode 实现
    const items: Array<{ title: string; scope: SftpWriteScope }> = [];
    if (confirmation.allowedScopes.includes('once')) items.push({ title: t('Allow once'), scope: 'once' });
    if (confirmation.allowedScopes.includes('directory')) items.push({ title: t('Allow this directory'), scope: 'directory' });
    if (confirmation.allowedScopes.includes('session')) items.push({ title: t('Allow this session'), scope: 'session' });

    const message = confirmation.stage === 'sensitive-double-check'
      ? t('Sensitive path modification requires double confirmation: {0}', confirmation.request.path)
      : t('Agent requested SFTP write to: {0}', confirmation.request.path);

    const picked = await vscode.window.showWarningMessage(message, { modal: true }, ...items.map(i => i.title));
    return items.find(i => i.title === picked)?.scope;
  }
}
```

```ts
import type { AgentConfirmationPort, CommandConfirmationRequest } from '../../agent/AgentConfirmationPort';
import type { SftpWriteConfirmation, SftpWriteScope } from '../../agent/SftpWriteAuthorizer';
import { formatRemoteCommandConfirmMessage } from '../../utils/commandPreview';

export interface DshApprovalRequest {
  agent: string;
  toolName: string;
  callId?: string;
  reason: string;
  displayReason?: string;
  signal?: AbortSignal;
}

export type DshApprovalOutcome = 'allowed-once' | 'rejected' | 'cancelled' | 'unavailable';

export interface DshApprovalService {
  request(req: DshApprovalRequest): Promise<DshApprovalOutcome>;
}

export interface DshCallContext {
  agent?: string;
  callId?: string;
  signal?: AbortSignal;
}

export class DshAgentConfirmation implements AgentConfirmationPort {
  constructor(
    private readonly options: {
      resolveApproval: () => DshApprovalService | undefined;
      getCurrentCallContext: () => DshCallContext | undefined;
    }
  ) {}

  async confirmCommand(request: CommandConfirmationRequest): Promise<boolean> {
    const approval = this.options.resolveApproval();
    const callCtx = this.options.getCurrentCallContext();

    if (!approval || !callCtx?.agent) {
      throw new Error(
        `Command "${request.command}" requires approval, but no human approval channel is available.`
      );
    }

    const displayReason = formatRemoteCommandConfirmMessage({
      serverLabel: request.serverLabel,
      host: request.host,
      command: request.command,
      destructive: request.destructive,
      riskSummaries: request.riskSummaries
    });

    const outcome = await approval.request({
      agent: callCtx.agent,
      toolName: 'run_remote_command',
      callId: callCtx.callId,
      reason: `Remote command requires confirmation: ${request.command}`,
      displayReason,
      signal: request.signal ?? callCtx.signal
    });

    return outcome === 'allowed-once';
  }

  async confirmSftpWrite(confirmation: SftpWriteConfirmation): Promise<SftpWriteScope | undefined> {
    const approval = this.options.resolveApproval();
    const callCtx = this.options.getCurrentCallContext();

    if (!approval || !callCtx?.agent) {
      throw new Error(`SFTP write to "${confirmation.request.path}" requires approval, but no approval channel is available.`);
    }

    const reason = confirmation.stage === 'sensitive-double-check'
      ? `Double confirmation required for sensitive path: ${confirmation.request.path}`
      : `SFTP write requested: ${confirmation.request.path}`;

    const outcome = await approval.request({
      agent: callCtx.agent,
      toolName: confirmation.request.operation,
      callId: callCtx.callId,
      reason,
      signal: callCtx.signal
    });

    return outcome === 'allowed-once' ? 'once' : undefined;
  }
}
```

```ts
import { AgentToolService } from '../../agent/AgentToolService';
import { RemoteCommandExecutor } from '../../agent/RemoteCommandExecutor';
import { SftpAgentService } from '../../agent/SftpAgentService';
import { SftpWriteAuthorizer } from '../../agent/SftpWriteAuthorizer';
import { ConfigManager } from '../../config/ConfigManager';
import type { Platform } from '../../core/ports';
import { SftpManager } from '../../sftp/SftpManager';
import { HostKeyStore } from '../../ssh/HostKeyStore';
import { TerminalContextRegistry } from '../../terminal/TerminalContext';
import { DshAgentConfirmation, type DshApprovalService, type DshCallContext } from './confirmation';

export interface DshAgentServiceOptions {
  readonly platform: Platform;
  readonly resolveApproval: () => DshApprovalService | undefined;
  readonly getCurrentCallContext: () => DshCallContext | undefined;
}

export function createDshAgentToolService(options: DshAgentServiceOptions): AgentToolService {
  const { platform } = options;
  const configManager = new ConfigManager(platform.state, platform.secrets);
  const hostKeyStore = new HostKeyStore(platform.state);
  const terminalContext = new TerminalContextRegistry();
  const confirmation = new DshAgentConfirmation({
    resolveApproval: options.resolveApproval,
    getCurrentCallContext: options.getCurrentCallContext
  });
  const executor = new RemoteCommandExecutor(configManager, hostKeyStore);
  const sftpManager = new SftpManager(configManager, hostKeyStore);
  const writeAuthorizer = new SftpWriteAuthorizer({
    confirm: (req) => confirmation.confirmSftpWrite(req)
  });
  const sftp = new SftpAgentService({
    sftpManager,
    configManager,
    terminalContext,
    authorizer: writeAuthorizer
  });

  return new AgentToolService({
    configManager,
    terminalContext,
    executor,
    sftp,
    confirmation
  });
}
```

```javascript
          stderr: { type: 'string' },

          truncated: { type: 'boolean' }
        },
        required: ['serverId', 'stdout', 'stderr', 'durationMs', 'timedOut', 'truncated']
      },
      render(_args, val) {
        return [
          {
            type: 'text',
            text: `[${val.serverId}] exit ${val.exitCode ?? 'signal'} (${val.durationMs}ms):\n\`\`\`\n${val.stdout}${val.stderr ? '\n[stderr]\n' + val.stderr : ''}\n\`\`\``
          }
        ];
      }
    },
    async execute(args, exec) {
      const service = resolveAgentService();
      return await service.runRemoteCommand({

        signal: exec?.signal
      });
    }
  };
}
```
